// сравниваем время
    int n = 500;

    DenseMatrix a(n, n);
    DenseMatrix b(n, n);

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            a.set_element(i, j, (i - j) % 10);
            b.set_element(i, j, (i + j + 10) % 10);
        }
    }

    auto start_naive = std::chrono::steady_clock::now();

    DenseMatrix c_naive = a.multiply_naive(b);

    auto end_naive = std::chrono::steady_clock::now();

    auto start_optimized = std::chrono::steady_clock::now();

    DenseMatrix c_optimized = a.multiply_optimized(b);

    auto end_optimized = std::chrono::steady_clock::now();

    double time_naive =
        std::chrono::duration<double>(end_naive - start_naive).count();

    double time_optimized =
        std::chrono::duration<double>(end_optimized - start_optimized).count();

    std::cout << "i-j-k: " << time_naive << " seconds\n";
    std::cout << "i-k-j: " << time_optimized << " seconds\n";

    bool equal = true;

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (c_naive.get_element(i, j) !=
                c_optimized.get_element(i, j)) {
                equal = false;
                }
        }
    }

    std::cout << "Results match: " << equal << '\n';
